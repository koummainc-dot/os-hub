function LineChartSVG({points,color,height=70}){
  if(!points||points.length<2) return <div style={{fontSize:10,color:"#94A3B8"}}>محتاج قياسين على الأقل لعرض الاتجاه بمرور الوقت</div>;

function DonutChart({data,size=120}){
  const total=data.reduce((s,d)=>s+d.value,0);
  if(total<=0) return <div style={{fontSize:10,color:"#94A3B8"}}>لا توجد بيانات كافية</div>;
  let acc=0;
  const r=size/2, cx=r, cy=r, sw=size*0.28;
  const segs=data.map(d=>{
    const frac=d.value/total;
    const start=acc*2*Math.PI - Math.PI/2; acc+=frac;
    const end=acc*2*Math.PI - Math.PI/2;
    const large=frac>0.5?1:0;
    const x1=cx+(r-sw/2)*Math.cos(start), y1=cy+(r-sw/2)*Math.sin(start);
    const x2=cx+(r-sw/2)*Math.cos(end), y2=cy+(r-sw/2)*Math.sin(end);
    return {d:`M ${x1} ${y1} A ${r-sw/2} ${r-sw/2} 0 ${large} 1 ${x2} ${y2}`,color:d.color,pct:Math.round(frac*100),label:d.label};
  });
  return(
    <div style={{display:"flex",alignItems:"center",gap:16}}>
      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
        {segs.map((s,i)=><path key={i} d={s.d} fill="none" stroke={s.color} strokeWidth={sw}/>)}
      </svg>
      <div style={{display:"flex",flexDirection:"column",gap:5}}>
        {segs.map((s,i)=>(
          <div key={i} style={{display:"flex",alignItems:"center",gap:6,fontSize:10}}>
            <span style={{width:9,height:9,borderRadius:2,background:s.color,flexShrink:0}}/>
            <span style={{color:"#57606A"}}>{s.label}</span>
            <span style={{fontWeight:700,color:"#16181D"}}>{s.pct}%</span>
          </div>
        ))}
      </div>
    </div>
  );
}
// Returns the most recent analytics entry within [from,to] (inclusive, either bound optional);
// falls back to overall latest entry when no date filter is set.
function entryInRange(rec,from,to){
  if(!rec||!rec.entries||rec.entries.length===0) return null;
  const filtered=rec.entries.filter(e=>(!from||e.date>=from)&&(!to||e.date<=to));
  return filtered.length?filtered[filtered.length-1]:(from||to?null:rec.entries[rec.entries.length-1]);
}

function dedupeInputs(kpis){
  const seen=new Map();
  (kpis||[]).forEach(k=>(k.inputs||[]).forEach(inp=>{ if(!seen.has(inp.id)) seen.set(inp.id,inp.label); }));
  return Array.from(seen.entries()).map(([id,label])=>({id,label}));
}

const KPI_TARGET_SCHEMA_VERSION=1;
function normalizeKpiTarget(k){
  const t=(k&&k.targetConfig&&typeof k.targetConfig==='object')?k.targetConfig:{};
  return {schemaVersion:t.schemaVersion||KPI_TARGET_SCHEMA_VERSION,mode:t.mode||'inherit',value:t.value!==undefined&&t.value!==''?Number(t.value):null,unit:t.unit||'',direction:t.direction||'higher_better',benchmarkType:t.benchmarkType||'none',benchmarkValue:t.benchmarkValue!==undefined&&t.benchmarkValue!==''?Number(t.benchmarkValue):null,scoreType:t.scoreType||'percent_of_target',cap:t.cap!==undefined&&t.cap!==''?Number(t.cap):100};
}
function kpiTargetValue(k){ const t=normalizeKpiTarget(k); return Number.isFinite(t.value)?t.value:null; }
function evaluateKpiPerformance(k,actual){
  const t=normalizeKpiTarget(k); const a=Number(actual);
  if(!Number.isFinite(a)||!Number.isFinite(t.value)||t.value===0) return null;
  const achieved=t.direction==='lower_better' ? (t.value/a)*100 : (a/t.value)*100;
  const score=t.cap>0?Math.min(t.cap,Math.max(0,achieved)):Math.max(0,achieved);
  let benchmark=null;
  if(Number.isFinite(t.benchmarkValue)){
    benchmark=t.direction==='lower_better' ? (t.benchmarkValue/a)*100 : (a/t.benchmarkValue)*100;
  }
  return {actual:a,target:t.value,achievement:achieved,score,benchmark,meetsTarget:t.direction==='lower_better'?a<=t.value:a>=t.value,direction:t.direction};
}
function kpiPerformanceLabel(result){ if(!result)return '—'; return `${result.score.toFixed(0)}%`; }

function kpiEntryScore(entry){
  const vals=Object.values(entry?.computed||{}).filter(v=>typeof v==="number"&&Number.isFinite(v));
  return vals.length?vals.reduce((a,b)=>a+b,0)/vals.length:null;
}
function kpiMetricTotals(records,key){
  let total=0,count=0;
  records.forEach(r=>{const v=parseAnalyticsNum(r._entry?.values?.[key]);if(v!==0){total+=v;count++;}else if(r._entry?.values?.[key]!==undefined&&r._entry?.values?.[key]!==""){count++;}});
  return {total,count,avg:count?total/count:0};
}
function kpiInteractions(records){

}

    {selectedAccount&&dashboard.enabled===false?<KpiEmpty title="لوحة Analytics متوقفة لهذا الحساب" desc="يمكن إعادة تفعيلها من إعدادات الحساب." icon="⏸️"/>:<div style={{background:"#fff",border:"1px solid rgba(0,0,0,0.07)",borderRadius:14,padding:8,marginBottom:14,display:"flex",gap:6,flexWrap:"wrap",alignItems:"stretch"}}>{tabs.map(t=><button key={t.id} onClick={()=>setTab(t.id)} style={{flex:"1 1 150px",textAlign:"right",padding:"10px 12px",borderRadius:10,border:`1px solid ${tab===t.id?A+"45":"transparent"}`,background:tab===t.id?A+"0F":"transparent",color:tab===t.id?A:"#57606A",cursor:"pointer",fontFamily:"inherit"}}><div style={{fontSize:11,fontWeight:800}}>{t.label}</div><div style={{fontSize:8,color:tab===t.id?A:"#94A3B8",marginTop:3}}>{t.desc}</div></button>)}</div>}
    <div style={{background:"#fff",border:"1px solid rgba(0,0,0,0.07)",borderRadius:14,padding:"10px 12px",marginBottom:16,display:"flex",alignItems:"center",gap:8,flexWrap:"wrap"}}><span style={{fontSize:10,fontWeight:700,color:"#57606A"}}>فلاتر التحليل</span><input type="date" value={dFrom} onChange={e=>setDFrom(e.target.value)} style={{...S.inp,margin:0,width:140,fontSize:10}}/><span style={{fontSize:9,color:"#94A3B8"}}>إلى</span><input type="date" value={dTo} onChange={e=>setDTo(e.target.value)} style={{...S.inp,margin:0,width:140,fontSize:10}}/>{accounts.length>0&&<select value={accId} onChange={e=>setAccId(e.target.value)} style={{...S.sel,margin:0,width:190,fontSize:10,color:"#000"}}><option value="">كل الحسابات</option>{accounts.map(a=><option key={a.id} value={a.id}>{a.name}</option>)}</select>}{(dFrom||dTo||accId)&&<button onClick={()=>{setDFrom("");setDTo("");setAccId("");}} style={{background:"none",border:"none",color:"#F87171",cursor:"pointer",fontFamily:"inherit",fontSize:10}}>× مسح الفلاتر</button>}<span style={{marginInlineStart:"auto",fontSize:9,color:"#94A3B8"}}>كل اللوحات تتبع نفس الفلاتر</span></div>
    {dashboard.enabled!==false&&<div>{tab==="entry"&&<KpiEntryTab plan={plan} setPlan={setPlan} accId={accId} dFrom={dFrom} dTo={dTo}/>} {tab==="format"&&<KpiFormatRollup plan={plan} accId={accId} dFrom={dFrom} dTo={dTo}/>} {tab==="platform"&&<KpiPlatformRollup plan={plan} setPlan={setPlan} accId={accId} dFrom={dFrom} dTo={dTo}/>} {tab==="system"&&<KpiSystemRollup plan={plan} accId={accId} dFrom={dFrom} dTo={dTo}/>} {tab==="layers"&&<KpiLayerIntelligence plan={plan} accId={accId} dFrom={dFrom} dTo={dTo}/>}</div>}
  </div>;
}
